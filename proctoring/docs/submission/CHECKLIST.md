```powershell
$candidateSha = 'ПОЛНЫЙ_SHA_ОТ_A01'
if ((git rev-parse HEAD).Trim() -ne $candidateSha) { throw 'Не тот кандидат' }
.\.venv\Scripts\python.exe demo\verify_demo.py
.\.venv\Scripts\python.exe demo\rehearse.py --mode synthetic --expected-sha $candidateSha --out demo\results\candidate_synthetic.json
```
